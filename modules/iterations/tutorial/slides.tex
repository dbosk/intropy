% beamer names its own example/definition/theorem blocks through the
% translator package, which ships no Swedish dictionary, so the titles
% come out in English.  Give it the Swedish words, before \usetheme.
\uselanguage{Swedish}
\languagepath{Swedish}
\deftranslation[to=Swedish]{Example}{Exempel}
\deftranslation[to=Swedish]{Definition}{Definition}
\deftranslation[to=Swedish]{Theorem}{Sats}
\deftranslation[to=Swedish]{Proof}{Bevis}
% translator's Swedish theorem dictionary calls a solution "Lösning",
% while didactic's own translation (which the notes job uses) says
% "Lösningsförslag".  The two jobs must agree, and this deck's solutions
% are proposals and not the only answers, so override the dictionary
% here --- after \uselanguage{Swedish}, which is what loads it.
\deftranslation[to=Swedish]{Solution}{Lösningsförslag}
\deftranslation[to=Swedish]{Solutions}{Lösningsförslagen}

% didactic provides exercise, remark, summary, example and definition for
% beamer itself, with Swedish titles, so no mapping is needed for them.
% `solution` is the exception: beamer declares it alongside theorem and
% lemma, so beamer got there first and didactic's \ProvideSemanticEnv is a
% no-op -- the body would be set in italics.  In this deck every solution
% is prose introducing a code chunk, so it must look like the other
% blocks; hence \DeclareDocumentEnvironment and not \Provide...
\DeclareDocumentEnvironment{solution}{o}{%
  \IfValueTF{#1}{%
    \begin{block}{\translate{Solution}: #1}
  }{%
    \begin{block}{\translate{Solution}}
  }
}{%
  \end{block}
}
