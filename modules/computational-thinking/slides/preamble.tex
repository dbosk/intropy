\usepackage[capitalize]{cleveref}
% cleveref ships no Swedish names, and for an unsupported babel main language
% it re-applies its English defaults at \begin{document} --- clobbering any
% \crefname set earlier in the preamble.  Register the Swedish names in an
% \AtBeginDocument hook (added after cleveref's own hook, so ours wins) to get
% "avsnitt"/"bilaga"/"tabell" instead of "Section"/"Appendix"/"Table".
\AtBeginDocument{%
  \crefname{chapter}{kapitel}{kapitel}%
  \Crefname{chapter}{Kapitel}{Kapitel}%
  \crefname{section}{avsnitt}{avsnitt}%
  \Crefname{section}{Avsnitt}{Avsnitt}%
  \crefname{subsection}{avsnitt}{avsnitt}%
  \Crefname{subsection}{Avsnitt}{Avsnitt}%
  \crefname{appendix}{bilaga}{bilagor}%
  \Crefname{appendix}{Bilaga}{Bilagor}%
  \crefname{figure}{figur}{figurer}%
  \Crefname{figure}{Figur}{Figurer}%
  \crefname{table}{tabell}{tabeller}%
  \Crefname{table}{Tabell}{Tabeller}%
  \crefname{equation}{ekvation}{ekvationer}%
  \Crefname{equation}{Ekvation}{Ekvationer}%
}
