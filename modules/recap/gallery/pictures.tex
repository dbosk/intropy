% Rebinding and mutating inside deposit: the state after each line.
% #1 the line (1, 2 or 3).  The numbers in the dicts are objects too, so
% that the second line shows the arrow in the shared dict moving from
% 1500 to 2001.  main's amount is listed first, and the objects stand at
% the heights of the names that point to them, so that no arrows cross.
\newcommand{\figRebind}[1]{%
  \begin{pyfig}
    \pyframe{m}{main}{amount: \pyptr, account: \pyptr}
    \pyframe[below=9mm of m.south west, anchor=north west]{dep}%
      {deposit}{account: \pyptr, amount: \pyptr}
    \pyval[at={($(m-amount.east)+(12mm,0)$)}, anchor=west]{v500}%
      {int}{500}
    \pydict[at={($(m-account.east)!0.5!(dep-account.east)+(12mm,0)$)},
            anchor=west]{d1}{"owner": "Ronja", "balance": \pyptr}
    \pyval[at={($(dep-amount.east)+(12mm,0)$)}, anchor=west]{v501}%
      {int}{501}
    \pyval[right=13mm of d1-1]{v1500}{int}{1500}
    \pyarrow{m-amount}{v500}
    \pyarrow{m-account}{d1}
    \pyarrow{dep-amount}{v501}
    \ifnumequal{#1}{1}{%
      \pyarrow{d1-1}{v1500}
    }{%
      \pyval[at={($(v1500.west |- m-account)$)}, anchor=west]{v2001}%
        {int}{2001}
      \pyarrow[out=0, in=180, looseness=1.3]{d1-1}{v2001}
    }%
    \ifnumequal{#1}{3}{%
      \pydict[at={($(v1500.east |- dep-account)+(5mm,0)$)}, anchor=west]%
        {d2}{"owner": "Pippi", "balance": \pyptr}
      \pyval[below=6mm of d2-1, pyfig type at=left]{v0}{int}{0}
      \pyarrow{d2-1}{v0}
      \pyarrow{dep-account}{d2.west |- dep-account}
    }{%
      \pyarrow{dep-account}{d1}
    }%
  \end{pyfig}%
}
