% contents.tex is woven from contents.nw; noweb.sty defines the chunk
% markup macros the weave emits.  The [minted] option makes the chunks
% render through minted (the weave runs tominted).  Frames have no real
% margin, so the sub-page tags go (nomargintag).
% The deck is code-dense; the notes' \footnotesize is too big here.
\setminted{fontsize=\scriptsize}

% The chunk cross-referencing apparatus (sub-page tags, defines/uses
% lists) is for the notes; on slides it is noise, and beamer overlays
% re-execute frame bodies so the sub-page labels would be multiply
% defined.  Gut it here; the notes keep it.
\def\sublabel#1{}
\def\subpageref#1{}
\def\nwindexdefn#1#2#3{}
\def\nwindexuse#1#2#3{}
\def\nwidentdefs#1{}
\def\nwidentuses#1{}
\def\nwused#1{}
\def\nwnotused#1{}
\def\nwalsodefined#1{}
% The "(used in ...)" tag on a definition line resolves a \subpageref, and
% the sub-page labels are gutted above, so it would leave undefined
% references behind.  Gut it too.

% PROJECT: language-specific.  beamer names its own example/definition/
% theorem blocks through the translator package, which ships no Swedish
% dictionary, so the titles come out in English.  Give it the words of the
% document's language, before \usetheme.  Delete this block for an English
% deck; translate the right-hand sides for another language.
\uselanguage{Swedish}
\languagepath{Swedish}
\deftranslation[to=Swedish]{Example}{Exempel}
\deftranslation[to=Swedish]{Definition}{Definition}
\deftranslation[to=Swedish]{Theorem}{Sats}
\deftranslation[to=Swedish]{Lemma}{Lemma}
\deftranslation[to=Swedish]{Corollary}{Följdsats}
\deftranslation[to=Swedish]{Proof}{Bevis}
% beamer defines "solution" itself, so didactic's \ProvideSemanticEnv
% leaves it alone and translator's own Swedish word ("Lösning") is used.
% The notes get didactic's word; say the same thing in both jobs.
\deftranslation[to=Swedish]{Solution}{Lösningsförslag}

\usetheme{Berlin}
% Berlin's navigation bar puts one box per frame in the headline, at
% beamer's default .1cm plus .05cm offset.  With the tenth section the
% bar grew wider than the slide: the last section's short title ran off
% the edge and every page logged an Overfull \hbox.  Make the boxes
% smaller, as in the other tutorial decks --- the bar is navigation, not
% content.  (Must come after \usetheme, which selects the mini frame
% template and with it the default size.)  Where a section's title is
% wider than its boxes, only a shorter short title helps: that is why
% the tenth section is [Texter].
\setbeamersize{mini frame size=.06cm, mini frame offset=.02cm}
\setbeamertemplate{footline}%{miniframes theme}
{%
  \begin{beamercolorbox}[colsep=1.5pt]{upper separation line foot}
  \end{beamercolorbox}
  \begin{beamercolorbox}[ht=2.5ex,dp=1.125ex,%
    leftskip=.3cm,rightskip=.3cm plus1fil]{author in head/foot}%
    \leavevmode{\usebeamerfont{author in head/foot}\insertshortauthor}%
    \hfill%
    {\usebeamerfont{institute in head/foot}%
     \usebeamercolor[fg]{institute in head/foot}\insertshortinstitute}%
  \end{beamercolorbox}%
  \begin{beamercolorbox}[ht=2.5ex,dp=1.125ex,%
    leftskip=.3cm,rightskip=.3cm plus1fil]{title in head/foot}%
    {\usebeamerfont{title in head/foot}\insertshorttitle}
    \hfill \insertframenumber%
  \end{beamercolorbox}%
  \begin{beamercolorbox}[colsep=1.5pt]{lower separation line foot}
  \end{beamercolorbox}
}
\setbeamercovered{transparent}
\setbeamertemplate{bibliography item}[text]

% Nine sections no longer fit on one table-of-contents frame at the
% normal size: beamer spaces the entries 1.5em apart, so a smaller font
% also shrinks the gaps.  With the tenth section \small ran off the frame
% as well, so the size is now \footnotesize.
\AtBeginSection[]{%
  \begin{frame}<beamer>
    \footnotesize
    \tableofcontents[currentsection]
  \end{frame}
}
% didactic installs an \AtBeginSubsection hook that emits a second table
% of contents per subsection.  This deck has one \subsection per section,
% all called Lösningsförslag, so the hook doubles the ToC frames and every
% second one says nothing the section's own added.  Silence it.
\AtBeginSubsection[]{}
