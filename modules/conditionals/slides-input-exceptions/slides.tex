% Seventy-odd slides in eleven sections: at the theme's default mini
% frame size the navigation bar comes out about a centimetre wider than
% the slide, which LaTeX reports as an Overfull \hbox on every page and
% which clips the last section's name.  Narrower dots with a smaller
% offset make the bar fit; nothing else in the theme changes.
\setbeamersize{mini frame size=.07cm, mini frame offset=.03cm}
\setbeamercovered{transparent}
\setbeamertemplate{bibliography item}[text]

% The deck has eleven sections, so the full table of contents no longer
% fits a frame at the theme's font size --- hence [shrink] on both hooks.
% The subsection hook overrides didactic's own (didactic.sty sets one
% too, without shrink), so that its frames fit as well.
\AtBeginSection[]{%
  \begin{frame}<beamer>[shrink]
    \tableofcontents[currentsection]
  \end{frame}
}
\AtBeginSubsection[]{%
  \begin{frame}<beamer>[shrink]
    \tableofcontents[currentsection,
                     subsectionstyle=show/shaded/hide,
                     subsubsectionstyle=show/show/hide]
  \end{frame}
}
