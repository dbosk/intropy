% contents.tex is woven from contents.nw; noweb.sty defines the chunk
% markup macros the weave emits.  The [minted] option makes the chunks
% render through minted (the weave runs tominted).  Frames have no real
% margin, so the sub-page tags go (nomargintag).
\usepackage[minted]{noweb}
\noweboptions{breakcode,nomargintag,noxref}
% The chunk cross-referencing apparatus (sub-page tags, defines/uses
% lists) is for the notes; on slides it is noise, and beamer overlays
% re-execute frame bodies so the sub-page labels would be multiply
% defined.  Gut it here; the notes keep it.
\def\sublabel#1{}
\def\subpageref#1{}
\def\nwindexdefn#1#2#3{}
\def\nwindexuse#1#2#3{}
\def\nwidentdefs#1{}
\def\nwidentuses#1{}
\def\nwused#1{}
\def\nwnotused#1{}
\def\nwalsodefined#1{}
% The "(used in ...)" tag on a definition line resolves a \subpageref, and
% the sub-page labels are gutted above, so it would leave undefined
% references behind.  Gut it too.
\def\nwusesondefline#1{}
\def\nwprevnextdefs#1#2{}

% PROJECT: language-specific.  beamer names its own example/definition/
% theorem blocks through the translator package, which ships no Swedish
% dictionary, so the titles come out in English.  Give it the words of the
% document's language, before \usetheme.  Delete this block for an English
% deck; translate the right-hand sides for another language.
\uselanguage{Swedish}
\languagepath{Swedish}
\deftranslation[to=Swedish]{Example}{Exempel}
\deftranslation[to=Swedish]{Definition}{Definition}
\deftranslation[to=Swedish]{Theorem}{Sats}
\deftranslation[to=Swedish]{Lemma}{Lemma}
\deftranslation[to=Swedish]{Corollary}{Följdsats}
\deftranslation[to=Swedish]{Proof}{Bevis}
